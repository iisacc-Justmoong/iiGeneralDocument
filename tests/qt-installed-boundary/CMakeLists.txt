cmake_minimum_required(VERSION 3.31)
project(DocumentInstalledQtBoundary LANGUAGES CXX)
set(CMAKE_DISABLE_FIND_PACKAGE_Qt6 TRUE)
set(CMAKE_DISABLE_FIND_PACKAGE_Qt5 TRUE)
set(CMAKE_DISABLE_FIND_PACKAGE_Qt TRUE)
add_library(Qt6::Core INTERFACE IMPORTED)
if(BOUNDARY_CASE STREQUAL "transitive")
    add_library(iiFileProvider::Native INTERFACE IMPORTED)
    target_link_libraries(iiFileProvider::Native INTERFACE Qt6::Core)
elseif(BOUNDARY_CASE STREQUAL "pdf")
    set(CMAKE_DISABLE_FIND_PACKAGE_iiXmlMarkup TRUE)
    set(CMAKE_DISABLE_FIND_PACKAGE_iiFileProviderNative TRUE)
endif()
if(BOUNDARY_CASE STREQUAL "pdf")
    find_package(iiGeneralDocument 0.4 CONFIG REQUIRED COMPONENTS Pdf)
    if(TARGET iiXml::Markup OR TARGET iiFileProvider::Native)
        message(FATAL_ERROR "PDF component imported an unrelated SDK")
    endif()
else()
    find_package(iiGeneralDocument 0.4 CONFIG REQUIRED)
endif()
