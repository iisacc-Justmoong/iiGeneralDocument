cmake_minimum_required(VERSION 3.31)
project(iiGeneralDocumentPdfConsumer LANGUAGES CXX)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_DISABLE_FIND_PACKAGE_Qt6 TRUE)
set(CMAKE_DISABLE_FIND_PACKAGE_Qt5 TRUE)
set(CMAKE_DISABLE_FIND_PACKAGE_Qt TRUE)
set(CMAKE_DISABLE_FIND_PACKAGE_iiFileProvider TRUE)
set(CMAKE_DISABLE_FIND_PACKAGE_iiXml TRUE)
set(CMAKE_DISABLE_FIND_PACKAGE_iiXmlMarkup TRUE)
set(CMAKE_DISABLE_FIND_PACKAGE_iiHtmlBlock TRUE)
find_package(iiGeneralDocument 0.3 CONFIG REQUIRED COMPONENTS Pdf)
foreach(_unwanted IN ITEMS Qt6::Core iiFileProvider::iiFileProvider
        iiXml::iiXml iiXml::Markup iiHtmlBlock::iiHtmlBlock iiGeneralDocument::iiGeneralDocument)
    if(TARGET ${_unwanted})
        message(FATAL_ERROR "The Pdf-only consumer loaded an unrelated package: ${_unwanted}")
    endif()
endforeach()
get_target_property(_links iiGeneralDocument::Pdf INTERFACE_LINK_LIBRARIES)
if(_links)
    message(FATAL_ERROR "Private PDF backends leaked into the public link interface: ${_links}")
endif()
add_executable(iiGeneralDocumentPdfConsumer main.cpp)
target_link_libraries(iiGeneralDocumentPdfConsumer PRIVATE iiGeneralDocument::Pdf)
set_target_properties(iiGeneralDocumentPdfConsumer PROPERTIES CXX_EXTENSIONS OFF)
