cmake_minimum_required(VERSION 3.31)
project(QtBoundaryFixture LANGUAGES NONE)
include("${IIGENERALDOCUMENT_SOURCE_DIR}/cmake/QtBoundary.cmake")
add_library(document INTERFACE)
add_library(provider INTERFACE)
add_library(native::provider ALIAS provider)
add_library(Qt6::Core INTERFACE IMPORTED)
# The presence of an unrelated Qt target must not prohibit a Qt-free SDK.
target_link_libraries(document INTERFACE "$<LINK_ONLY:native::provider>")
if(BOUNDARY_CASE STREQUAL "transitive")
    target_link_libraries(provider INTERFACE "$<LINK_ONLY:Qt6::Core>")
elseif(BOUNDARY_CASE STREQUAL "qt_alias")
    add_library(disguised_qt ALIAS Qt6::Core)
    target_link_libraries(provider INTERFACE disguised_qt)
elseif(BOUNDARY_CASE STREQUAL "direct")
    target_link_libraries(document INTERFACE Qt6::Core)
elseif(BOUNDARY_CASE STREQUAL "library_path")
    add_library(opaque_sdk UNKNOWN IMPORTED)
    set_target_properties(opaque_sdk PROPERTIES IMPORTED_LOCATION "/somewhere/libQt6Core.so")
    target_link_libraries(provider INTERFACE opaque_sdk)
elseif(BOUNDARY_CASE STREQUAL "windows_library_path")
    add_library(opaque_sdk UNKNOWN IMPORTED)
    set_target_properties(opaque_sdk PROPERTIES IMPORTED_LOCATION "C:/SDK/Qt6Samples/Qt6Core.dll")
    target_link_libraries(provider INTERFACE opaque_sdk)
elseif(BOUNDARY_CASE STREQUAL "include_path")
    set_target_properties(provider PROPERTIES INTERFACE_INCLUDE_DIRECTORIES "/some/Qt6Core/include")
elseif(BOUNDARY_CASE STREQUAL "compile_definition")
    target_compile_definitions(provider INTERFACE QT_CORE_LIB)
elseif(BOUNDARY_CASE STREQUAL "link_option")
    target_link_options(provider INTERFACE -lQt6Gui)
elseif(BOUNDARY_CASE STREQUAL "qt5")
    add_library(Qt5::Xml INTERFACE IMPORTED)
    target_link_libraries(provider INTERFACE Qt5::Xml)
elseif(BOUNDARY_CASE STREQUAL "qt_bluetooth")
    target_link_options(provider INTERFACE -lQt6Bluetooth)
elseif(BOUNDARY_CASE STREQUAL "qt_feature_definition")
    target_compile_definitions(provider INTERFACE QT_BLUETOOTH_LIB)
elseif(BOUNDARY_CASE STREQUAL "qt_monolithic")
    target_link_options(provider INTERFACE -lQt)
elseif(BOUNDARY_CASE STREQUAL "safe_qt_named_path")
    file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/Qt6Samples/include")
    set_target_properties(provider PROPERTIES
        INTERFACE_INCLUDE_DIRECTORIES "${CMAKE_CURRENT_BINARY_DIR}/Qt6Samples/include")
    target_link_options(provider INTERFACE "-Wl,-rpath,/some/Qt6Samples/lib")
elseif(BOUNDARY_CASE STREQUAL "safe")
    add_library(system_threads INTERFACE)
    target_link_libraries(provider INTERFACE system_threads)
else()
    message(FATAL_ERROR "Unknown test case")
endif()
iigeneraldocument_require_qt_free(document)
